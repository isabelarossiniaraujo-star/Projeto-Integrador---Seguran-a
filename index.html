// Seleciona os elementos do HTML const form = document.getElementById('form-registro'); const inputNome = document.getElementById('nome'); const selectCategoria = document.getElementById('categoria'); const muralCards = document.getElementById('mural-cards'); if (!form || !inputNome || !selectCategoria || !muralCards) { throw new Error('Elemento não encontrado: confira os IDs no HTML.'); } const STORAGE_KEY = 'registros'; // Carrega os registros salvos com segurança function carregarRegistros() { try { const dados = JSON.parse(localStorage.getItem(STORAGE_KEY)); return Array.isArray(dados) ? dados : []; } catch { return []; } } function salvarRegistros() { try { localStorage.setItem(STORAGE_KEY, JSON.stringify(registros)); } catch (erro) { console.error('Não foi possível salvar:', erro); } } let registros = carregarRegistros(); // Cria um cartão sem usar innerHTML (evita XSS) function criarCard(registro, indice) { const card = document.createElement('div'); card.classList.add('card'); const titulo = document.createElement('h3'); titulo.textContent = registro.nome; const categoria = document.createElement('p'); const rotulo = document.createElement('strong'); rotulo.textContent = 'Categoria: '; categoria.append(rotulo, registro.categoria); const btnExcluir = document.createElement('button'); btnExcluir.type = 'button'; btnExcluir.textContent = 'Excluir'; btnExcluir.addEventListener('click', () => removerRegistro(indice)); card.append(titulo, categoria, btnExcluir); return card; } // Renderiza os cartões no mural function atualizarMural() { muralCards.replaceChildren( ...registros.map((registro, indice) => criarCard(registro, indice)) ); } function removerRegistro(indice) { registros.splice(indice, 1); salvarRegistros(); atualizarMural(); } // Envio do formulário form.addEventListener('submit', (event) => { event.preventDefault(); const nome = inputNome.value.trim(); if (!nome) { inputNome.focus(); return; } registros.push({ nome, categoria: selectCategoria.value }); salvarRegistros(); atualizarMural(); form.reset(); inputNome.focus(); }); // Exibe os cartões ao carregar a página atualizarMural();
